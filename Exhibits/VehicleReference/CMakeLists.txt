cmake_minimum_required(VERSION 3.20)
project(VehicleReference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(Root "../.." ABSOLUTE)
set(Rendering "${Root}/Frontier/Engine/Rendering")
set(ReferenceSources
    ${Root}/Frontier/Engine/Diagnostics/LogRingStore.cpp
    ${Root}/Frontier/Engine/Diagnostics/LogCategories.cpp
    ${Root}/Frontier/Engine/Diagnostics/LogStream.cpp)
foreach(Source ClusterNesting VisibilityCull DiscrepancySequence ReservoirSampling VarianceFiltering
               DistanceField SurfaceCards AtmosphereModel CloudField FogMedium ReferenceRenderer)
    list(APPEND ReferenceSources "${Rendering}/${Source}.cpp")
endforeach()
find_package(Threads REQUIRED)
add_executable(VehicleCapture Code/VehicleCapture.cpp ${ReferenceSources})
add_executable(TriangleVisibilityValidation Code/TriangleVisibilityValidation.cpp ${ReferenceSources})
add_executable(TriangleDistanceValidation Code/TriangleDistanceValidation.cpp ${ReferenceSources})
add_executable(LightValidation ../ClusteredLighting/Code/LightValidation.cpp ${ReferenceSources})
foreach(Target VehicleCapture TriangleVisibilityValidation TriangleDistanceValidation LightValidation)
    target_include_directories(${Target} PRIVATE "${Root}/Frontier")
    target_link_libraries(${Target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${Target} PRIVATE /utf-8)
    else()
        target_compile_options(${Target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME LightValidation COMMAND LightValidation)
add_test(NAME TriangleVisibilityValidation COMMAND TriangleVisibilityValidation)
add_test(NAME TriangleDistanceValidation COMMAND TriangleDistanceValidation)

add_subdirectory("${Root}/Exhibits/CompleteRenderer" "${CMAKE_CURRENT_BINARY_DIR}/complete")
