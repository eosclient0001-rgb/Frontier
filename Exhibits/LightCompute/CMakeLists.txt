cmake_minimum_required(VERSION 3.20)
project(LightCompute LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(Root "../.." ABSOLUTE)
find_path(VulkanHeaders vulkan/vulkan.h
    HINTS "${Root}/ExternalPackages/VulkanCompute/Vulkan-Headers-1.4.321/include" "$ENV{VULKAN_SDK}/include"
    REQUIRED)
set(Sources)
foreach(Name ClusterNesting VisibilityCull DiscrepancySequence ReservoirSampling VarianceFiltering
             DistanceField SurfaceCards AtmosphereModel CloudField FogMedium ReferenceRenderer)
    list(APPEND Sources "${Root}/Frontier/Engine/Rendering/${Name}.cpp")
endforeach()
foreach(Name LogRingStore LogCategories LogStream)
    list(APPEND Sources "${Root}/Frontier/Engine/Diagnostics/${Name}.cpp")
endforeach()
add_executable(LightComputeValidation Code/LightComputeValidation.cpp ${Sources})
# 📝 A generic build can omit the optional generated fixture; preparing it later must force recompilation.
set(MaterialOffsets "${Root}/_ClaudeScratch/build/light-compute/MaterialSceneOffsets.h")
if(EXISTS "${MaterialOffsets}")
    set_source_files_properties(Code/LightComputeValidation.cpp PROPERTIES OBJECT_DEPENDS "${MaterialOffsets}")
endif()
target_include_directories(LightComputeValidation PRIVATE "${Root}/Frontier" "${Root}" "${VulkanHeaders}")
find_package(Threads REQUIRED)
target_link_libraries(LightComputeValidation PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(LightComputeValidation PRIVATE /utf-8 /W4)
else()
    target_compile_options(LightComputeValidation PRIVATE -Wall -Wextra -Werror)
endif()
