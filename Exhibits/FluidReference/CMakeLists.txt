cmake_minimum_required(VERSION 3.20)
project(PlumeLatticeValidation LANGUAGES CXX)
option(FRONTIER_PLUME_SANITIZERS "Instrument the native footprint validation" OFF)
get_filename_component(Root "../.." ABSOLUTE)
add_subdirectory("${Root}/Frontier/Engine/Rendering" "${CMAKE_CURRENT_BINARY_DIR}/rendering-engine")
add_executable(PlumeLatticeFootprintValidation Code/PlumeLatticeFootprintValidation.cpp)
target_link_libraries(PlumeLatticeFootprintValidation PRIVATE FrontierPlumeLatticeFootprint)
target_compile_features(PlumeLatticeFootprintValidation PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(PlumeLatticeFootprintValidation PRIVATE /utf-8 /W4)
else()
    target_compile_options(PlumeLatticeFootprintValidation PRIVATE -Wall -Wextra -Werror)
endif()
if(FRONTIER_PLUME_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "This validation configures GCC/Clang ASan and UBSan")
    endif()
    foreach(Target FrontierPlumeLatticeFootprint PlumeLatticeFootprintValidation)
        target_compile_options(${Target} PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
    endforeach()
    target_link_options(PlumeLatticeFootprintValidation PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME PlumeLatticeFootprintGuards COMMAND PlumeLatticeFootprintValidation --guards)
